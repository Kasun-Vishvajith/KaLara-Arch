cmake_minimum_required(VERSION 3.25)
project(KaLaraArch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KALARA_BUILD_EDITOR "Build native Qt desktop editor" ON)
include(CTest)
add_library(kalara_core core/diagnostic.cpp core/geometry.cpp)
target_include_directories(kalara_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE kalara_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE kalara_core)
  add_test(NAME geometry_tests COMMAND geometry_tests)
endif()
if(KALARA_BUILD_EDITOR)
  find_package(Qt6 6.7.2 EXACT REQUIRED COMPONENTS Widgets Test)
  set(CMAKE_AUTOMOC ON)
  add_library(kalara_editor editor/action_registry.cpp editor/main_window.cpp editor/session.cpp)
  target_link_libraries(kalara_editor PUBLIC kalara_core Qt6::Widgets)
  add_executable(kalara_arch WIN32 editor/main.cpp)
  target_link_libraries(kalara_arch PRIVATE kalara_editor)
  if(BUILD_TESTING)
    add_executable(shell_tests tests/shell_tests.cpp)
    target_link_libraries(shell_tests PRIVATE kalara_editor Qt6::Test)
    add_test(NAME shell_tests COMMAND shell_tests)
    set_tests_properties(shell_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 45)
  endif()
endif()
