cmake_minimum_required(VERSION 3.25)
project(KaLaraArch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KALARA_BUILD_EDITOR "Build native Qt desktop editor" ON)
include(CTest)
add_library(kalara_core core/diagnostic.cpp core/geometry.cpp)
target_include_directories(kalara_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(kalara_architecture architecture/model.cpp)
target_include_directories(kalara_architecture PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(kalara_architecture PUBLIC kalara_core)
add_library(kalara_runtime runtime/command_service.cpp architecture/wall_authoring.cpp)
target_include_directories(kalara_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(kalara_runtime PUBLIC kalara_architecture)
add_library(kalara_render render/scene.cpp render/camera.cpp render/frame_recorder.cpp)
target_include_directories(kalara_render PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(kalara_render PUBLIC kalara_architecture)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE kalara_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE kalara_core)
  add_test(NAME geometry_tests COMMAND geometry_tests)
  add_executable(command_service_tests tests/command_service_tests.cpp)
  target_link_libraries(command_service_tests PRIVATE kalara_runtime)
  add_test(NAME command_service_tests COMMAND command_service_tests)
  add_executable(wall_authoring_tests tests/wall_authoring_tests.cpp)
  target_link_libraries(wall_authoring_tests PRIVATE kalara_runtime)
  add_test(NAME wall_authoring_tests COMMAND wall_authoring_tests)
endif()
if(KALARA_BUILD_EDITOR)
  find_package(Qt6 6.7.2 EXACT REQUIRED COMPONENTS Core Widgets OpenGLWidgets Test)
  set(CMAKE_AUTOMOC ON)
  add_library(kalara_persistence persistence/json_codec.cpp persistence/migration.cpp)
  target_include_directories(kalara_persistence PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(kalara_persistence PUBLIC kalara_architecture Qt6::Core)
  add_library(kalara_editor editor/action_registry.cpp editor/main_window.cpp editor/session.cpp editor/plan_viewport.cpp editor/input_system.cpp)
  target_link_libraries(kalara_editor PUBLIC kalara_core kalara_runtime kalara_render Qt6::Widgets Qt6::OpenGLWidgets)
  add_executable(kalara_arch WIN32 editor/main.cpp)
  target_link_libraries(kalara_arch PRIVATE kalara_editor)
  if(BUILD_TESTING)
    add_executable(model_codec_tests tests/model_codec_tests.cpp)
    target_link_libraries(model_codec_tests PRIVATE kalara_persistence)
    target_compile_definitions(model_codec_tests PRIVATE KALARA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME model_codec_tests COMMAND model_codec_tests)
    set_tests_properties(model_codec_tests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    add_executable(shell_tests tests/shell_tests.cpp)
    target_link_libraries(shell_tests PRIVATE kalara_editor Qt6::Test)
    add_test(NAME shell_tests COMMAND shell_tests)
    set_tests_properties(shell_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 45)
    add_executable(render_tests tests/render_tests.cpp)
    target_link_libraries(render_tests PRIVATE kalara_editor Qt6::Test)
    target_compile_definitions(render_tests PRIVATE KALARA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME render_tests COMMAND render_tests)
    set_tests_properties(render_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 45)
    add_executable(input_system_tests tests/input_system_tests.cpp)
    target_link_libraries(input_system_tests PRIVATE kalara_editor)
    add_test(NAME input_system_tests COMMAND input_system_tests)
    set_tests_properties(input_system_tests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  endif()
endif()
